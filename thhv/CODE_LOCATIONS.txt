                  u64 a0, u64 a1, u64 a2, u64 a3, u64 a4,
                  u64 *r0, u64 *r1, u64 *r2)
    {
        u64 status, o0, o1, o2;
        register u64 _a4 asm("r8") = a4;
        asm volatile("vmcall"
            : "=a"(status), "=D"(o0), "=S"(o1), "=d"(o2)
            : "a"(opcode), "D"(a0), "S"(a1), "d"(a2),
              "c"(a3), "r"(_a4)
            : "r9", "r10", "r11", "memory", "cc"
        );
        if (r0) *r0 = o0;
        if (r1) *r1 = o1;
        if (r2) *r2 = o2;
        return status;
    }
