    [async]:  wait_queue_head_t exit_wq;         // VP exit parking (incomplete!)

  struct thhv_irqfd {

    __u32 rsvd;
  }

  struct thhv_ioeventfd {
    __s32 fd;           // User-provided eventfd

    __u32 len;          // 1, 2, or 4 bytes
    __u32 datamatch;    // Optional pattern to match
  }
